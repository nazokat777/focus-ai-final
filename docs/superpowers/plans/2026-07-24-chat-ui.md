# Chat UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Do'st / ustoz / shogird bilan 1-ga-1 yozishma interfeysi qurish (backend `send_msg`/`get_msgs` RPC allaqachon tayyor).

**Architecture:** Bitta `www/index.html` fayliga: yangi `#chat` `.detail` paneli + CSS, chat dvigateli (ochish/render/polling/yuborish), do'stlar panelidagi qatorlardan kirish nuqtalari, shogird (`wards()`) seksiyasi, o'qilmagan nuqtalari. Backend o'zgarmaydi. Jonlilik = polling (3s). Faqat bulut rejimida.

**Tech Stack:** Vanilla ES5 JS, WebAudio, localStorage, Supabase-js (CDN, mavjud). Framework/build yo'q. Test: Playwright (`_*.cjs`, gitignored) + manba-ajratib unit test.

```js
/* ============ CHAT (do'st/ustoz/shogird bilan; polling) ============ */
var chatWith=null, chatTimer=null, chatLastLen=-1;
function chatSeenKey(code){return 'focusai.chat.seen.'+code;}
function chatMarkSeen(code,iso){try{localStorage.setItem(chatSeenKey(code),iso||new Date().toISOString());}catch(e){}}
function chatSeenAt(code){try{return localStorage.getItem(chatSeenKey(code))||'';}catch(e){return '';}}
function chatIsUnread(msgs,me,seen){
  var last='';
  for(var i=0;i<msgs.length;i++){if(msgs[i].from_code!==me&&msgs[i].created_at>last)last=msgs[i].created_at;}
  return !!last&&(!seen||last>seen);
}
function chatUnread(code){
  if(!Social.cloud())return Promise.resolve(false);
  return Social.getMessages(code).then(function(msgs){
    return chatIsUnread(msgs||[],Social.myCode(),chatSeenAt(code));
  },function(){return false;});
}
function renderChat(msgs){
  var body=$('#chatBody');if(!body)return;
  var me=Social.myCode();
  if(!msgs||!msgs.length){body.innerHTML='<div class="chat-empty">'+T('chat_empty')+'</div>';return;}
  body.innerHTML=msgs.map(function(m){
    return '<div class="chat-msg '+(m.from_code===me?'me':'them')+'">'+esc(m.body)+'</div>';
  }).join('');
  body.scrollTop=body.scrollHeight;
}
function chatPoll(){
  if(!chatWith)return;
  var code=chatWith.code;
  Social.getMessages(code).then(function(msgs){
    if(!chatWith||chatWith.code!==code)return;
    if(msgs.length!==chatLastLen){
      chatLastLen=msgs.length;renderChat(msgs);
      if(msgs.length)chatMarkSeen(code,msgs[msgs.length-1].created_at);
    }
  });
}
function openChat(code,name,kind){
  if(!Social.cloud()){toast(T('chat_cloud_only'));return;}
  chatWith={code:code,name:name,kind:kind};chatLastLen=-1;
  $('#chatName').textContent=name||code;
  $('#chatRole').textContent=code;
  $('#chatInput').placeholder=T('chat_ph');$('#chatInput').value='';
  $('#chatBody').innerHTML='<div class="chat-empty">…</div>';
  $('#chat').classList.add('open');sTap();
  chatPoll();
  if(chatTimer)clearInterval(chatTimer);
  chatTimer=setInterval(chatPoll,3000);
}
function closeChat(){
  if(chatTimer){clearInterval(chatTimer);chatTimer=null;}
  chatWith=null;$('#chat').classList.remove('open');sTap();
  if($('#friends')&&$('#friends').classList.contains('open'))renderFriends();
}
function chatSend(){
  if(!chatWith)return;
  var inp=$('#chatInput'),body=(inp.value||'').trim().slice(0,500);
  if(!body)return;
  inp.value='';
  Social.sendMessage(chatWith.code,body).then(function(ok){
    if(ok){chatLastLen=-1;chatPoll();buzz(8);}
    else{toast(T('chat_err'));inp.value=body;}
  },function(){toast(T('chat_err'));inp.value=body;});
}
(function(){
  var cc=$('#chatClose');if(cc)cc.addEventListener('click',closeChat);
  var cs=$('#chatSend');if(cs)cs.addEventListener('click',chatSend);
  var ci=$('#chatInput');if(ci)ci.addEventListener('keydown',function(e){if(e.key==='Enter'){e.preventDefault();chatSend();}});
})();
```
